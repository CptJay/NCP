
### Matrices ###
Matrix m1 = [2 5 | 3 9];
Matrix m2 = [2 7 | 3 4];

#Addition
Matrix m3 = m1 + m2; #Oppassen voor dimensie!

#Substraction
Matrix m4 = m3 - m2; #Oppassen voor dimensie!

#Multiplication
m1 = m4 * m2;

#Transpose
Matrix m4 = Tra(m2); #m4 = [1 3 | 2 4]

#Determinant
double f = Det(m4);

#Inverse
Matrix g = Inv(m4);


### Polynomials ###

Poly f(x) = 2x^2 + 1x^1 + 9;
